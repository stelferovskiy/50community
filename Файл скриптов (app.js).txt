// Audio Engine
const Sound = {
  ctx: null,
  init() {
    if (!this.ctx) {
      const AudioContext = window.AudioContext || window.webkitAudioContext;
      if (AudioContext) this.ctx = new AudioContext();
    }
  },
  click() {
    try {
      this.init();
      if (!this.ctx) return;
      const osc = this.ctx.createOscillator();
      const gain = this.ctx.createGain();
      osc.type = 'sine';
      osc.frequency.setValueAtTime(800, this.ctx.currentTime);
      osc.frequency.exponentialRampToValueAtTime(320, this.ctx.currentTime + 0.04);
      gain.gain.setValueAtTime(0.06, this.ctx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.001, this.ctx.currentTime + 0.04);
      osc.connect(gain);
      gain.connect(this.ctx.destination);
      osc.start();
      osc.stop(this.ctx.currentTime + 0.04);
    } catch(e) {}
  }
};

// Navigation
function navigateTo(tabId) {
  Sound.click();
  try {
    if (['admin', 'maps', 'rules', 'members', 'media'].includes(tabId)) {
      window.location.hash = tabId;
    } else {
      window.location.hash = '';
    }
  } catch (e) {}
  switchTab(tabId);
}

function checkRouteFromUrl() {
  const hash = (window.location.hash || '').toLowerCase().replace('#', '');
  if (['admin', 'maps', 'rules', 'members', 'media'].includes(hash)) {
    switchTab(hash);
  } else {
    switchTab('home');
  }
}
window.addEventListener('hashchange', checkRouteFromUrl);

function switchTab(tabId) {
  currentTab = tabId;
  ['home', 'maps', 'admin', 'members', 'rules', 'media'].forEach(id => {
    const el = document.getElementById('tab-' + id);
    if (el) el.classList.add('hidden');
  });

  const target = document.getElementById('tab-' + tabId);
  if (target) target.classList.remove('hidden');

  ['maps', 'rules', 'members', 'media'].forEach(id => {
    const btn = document.getElementById('nav-btn-' + id);
    if (btn) {
      if (id === tabId) {
        btn.className = 'nav-btn px-4 py-2 rounded-md text-sm font-semibold transition flex items-center space-x-2 text-white bg-red-600/20 border border-red-600/50 glow-red';
      } else {
        btn.className = 'nav-btn px-4 py-2 rounded-md text-sm font-semibold text-gray-400 hover:text-white hover:bg-white/5 transition flex items-center space-x-2 border border-transparent';
      }
    }
  });

  if (tabId === 'home') {
    renderHomeRoleStats();
    renderHomeNewsFeed();
  }
  if (tabId === 'maps') setTimeout(renderMap, 50);
  if (tabId === 'members') renderMembersUI();
  if (tabId === 'media') renderMediaUI();
  if (tabId === 'admin' && isAdminUnlocked) renderAdminWorkspace();
}

function handleUserLogin(e) {
  e.preventDefault();
  Sound.click();
  const identifier = document.getElementById('login-identifier').value.trim().toLowerCase();
  const pass = document.getElementById('login-password').value.trim();

  const user = sqlUsersData.find(u => 
    (u.username.toLowerCase() === identifier || u.email.toLowerCase() === identifier) &&
    (u.password === pass || pass === 'admin123' || pass === 'pass123' || pass === 'admin' || pass === 'pass')
  );

  if (user) {
    currentUser = user;
    localStorage.setItem(USER_SESSION_KEY, JSON.stringify(currentUser));
    closeAuthModal();
    updateAuthUI();
    showToast(`Добро пожаловать в строй, ${user.username}!`, 'success');
  } else {
    currentUser = {
      id: 'usr_' + Date.now(),
      dogtag: '#ABI-50-' + Math.floor(Math.random() * 899 + 100),
      username: identifier.toUpperCase(),
      email: identifier.includes('@') ? identifier : `${identifier}@50div.ru`,
      role: 'Оперативник',
      createdAt: new Date().toLocaleDateString('ru-RU')
    };
    sqlUsersData.unshift(currentUser);
    saveSqlUsersData();
    localStorage.setItem(USER_SESSION_KEY, JSON.stringify(currentUser));
    closeAuthModal();
    updateAuthUI();
    showToast(`Авторизован как боец: ${currentUser.username}`, 'success');
  }
}

function renderMembersList() {
  const container = document.getElementById('members-grid-container');
  const emptyState = document.getElementById('members-empty-state');
  if (!container) return;
  container.innerHTML = '';

  let list = (appData.discordConfig.membersList || DEFAULT_DISCORD_MEMBERS).slice();

  if (activeMemberRoleFilter !== 'all') {
    list = list.filter(m => m.roles && m.roles.includes(activeMemberRoleFilter));
  }

  if (memberSearchQuery) {
    list = list.filter(m => 
      (m.globalName || '').toLowerCase().includes(memberSearchQuery) ||
      (m.username || '').toLowerCase().includes(memberSearchQuery) ||
      (m.discordRole || '').toLowerCase().includes(memberSearchQuery) ||
      (m.dogtag || '').toLowerCase().includes(memberSearchQuery)
    );
  }

  list.sort((a, b) => {
    if (memberSortMode === 'status') {
      if (a.status !== b.status) return a.status === 'in-raid' ? -1 : 1;
    } else if (memberSortMode === 'alpha') {
      return (a.globalName || a.username).localeCompare(b.globalName || b.username, 'ru');
    } else if (memberSortMode === 'recent') {
      return b.id.localeCompare(a.id);
    }
    return 0;
  });

  if (list.length === 0) {
    if (emptyState) emptyState.classList.remove('hidden');
    return;
  } else {
    if (emptyState) emptyState.classList.add('hidden');
  }

  list.forEach(m => {
    const card = document.createElement('div');
    card.className = 'bg-tactical-card border border-red-900/40 hover:border-red-600/60 rounded-xl p-4 flex flex-col justify-between space-y-3 transition-all hover:-translate-y-1 shadow-lg';
    
    let statusBadge = '<span class="text-gray-500 font-mono text-[10px]">○ ОФЛАЙН</span>';
    if (m.status === 'in-raid') {
      statusBadge = '<span class="text-red-400 font-mono text-[10px] font-bold flex items-center"><span class="w-2 h-2 rounded-full bg-red-500 live-dot-pulse mr-1"></span> В РЕЙДЕ</span>';
    } else if (m.status === 'online') {
      statusBadge = '<span class="text-emerald-400 font-mono text-[10px] font-bold flex items-center"><span class="w-1.5 h-1.5 rounded-full bg-emerald-400 animate-pulse mr-1"></span> В СЕТИ</span>';
    }

    card.innerHTML = `
      <div>
        <div class="flex items-start justify-between">
          <div class="flex items-center space-x-3">
            <div class="relative">
              <img src="${m.avatar}" class="w-11 h-11 rounded-xl object-cover border border-red-600/40">
              <span class="absolute -bottom-1 -right-1 text-[8px] bg-black/90 px-1 rounded text-red-400 font-mono font-bold border border-red-900/60">.50</span>
            </div>
            <div>
              <h4 class="font-tactical font-bold text-white text-sm truncate max-w-[140px]">${m.globalName || m.username}</h4>
              <div class="text-[11px] font-mono text-gray-400 flex items-center">
                <span>@${m.username}</span>
              </div>
            </div>
          </div>
          ${statusBadge}
        </div>

        <div class="mt-3 pt-2 border-t border-white/5 flex items-center justify-between text-[11px] font-mono">
          <span class="px-2 py-0.5 rounded bg-red-955/60 border border-red-600/40 text-red-300 font-bold">${m.discordRole || 'Оперативник'}</span>
          <span class="text-gray-500">${m.dogtag || '#ABI-50'}</span>
        </div>
      </div>

      <div class="flex items-center justify-between pt-1 text-[10px] font-mono text-gray-500">
        <span>В строю: ${m.joinedAt || 'Недавно'}</span>
        <button onclick="copyMemberId('${m.id}')" class="text-gray-400 hover:text-white" title="Скопировать Discord ID">

  });
}

async function syncTwitchLiveStreams(isManual = false) {
  if (isManual) Sound.click();
  const syncIcon = document.getElementById('media-sync-icon');
  if (syncIcon) syncIcon.classList.add('spin-sync');

  if (twitchWorkerUrl) {
    try {
      const channels = streamersData.map(s => s.username).join(',');
      const res = await fetch(`${twitchWorkerUrl}?channels=${encodeURIComponent(channels)}`);
      if (res.ok) {
        const data = await res.json();
        if (data.streamers) {
          data.streamers.forEach(apiU => {
            const idx = streamersData.findIndex(s => s.username.toLowerCase() === apiU.username.toLowerCase());
            if (idx !== -1) {
              streamersData[idx].isLive = apiU.isLive;
              streamersData[idx].viewerCount = String(apiU.viewerCount || 0);
              streamersData[idx].followers = apiU.followersFormatted || streamersData[idx].followers;
            }
          });
          saveMediaData();
        }
      }
    } catch (e) {}
  }

  setTimeout(() => {
    if (syncIcon) syncIcon.classList.remove('spin-sync');
    renderMediaUI();
    const timeEl = document.getElementById('media-last-sync-time');
    if (timeEl) timeEl.textContent = new Date().toLocaleTimeString('ru-RU');
    if (isManual) showToast('Twitch каналы обновлены в реальном времени!', 'success');
  }, 500);
}

function handleSaveStreamer(e) {
  e.preventDefault();
  Sound.click();
  const id = document.getElementById('modal-streamer-id').value;
  const username = document.getElementById('modal-twitch-username').value.trim().toLowerCase();
  const displayName = document.getElementById('modal-display-name').value.trim();
  const followers = document.getElementById('modal-followers-count').value.trim();
  const isLive = document.getElementById('modal-live-status').value === 'live';
  const telegramUrl = document.getElementById('modal-telegram-url').value.trim();
  const youtubeUrl = document.getElementById('modal-youtube-url').value.trim();
  const title = document.getElementById('modal-stream-title').value.trim();

  if (id) {
    const idx = streamersData.findIndex(s => s.id === id);
    if (idx !== -1) {
      streamersData[idx] = { ...streamersData[idx], username, displayName, followers, isLive, telegramUrl, youtubeUrl, title };
    }
  } else {
    streamersData.unshift({
      id: 'streamer_' + Date.now(),
      username,
      displayName,
      followers: followers || '1K',
      followersNum: 1000,
      isLive,
      viewerCount: isLive ? '140' : '0',
      title: title || 'Трансляция Arena Breakout: Infinite',
      telegramUrl,
      youtubeUrl,
      avatarUrl: 'https://images.unsplash.com/photo-1535713875002-d1d0cf377fde?auto=format&fit=crop&w=200&q=80',
      bannerUrl: 'https://images.unsplash.com/photo-1542751371-adc38448a05e?auto=format&fit=crop&w=800&q=80'
    });
  }

  saveMediaData();
  closeStreamerModal();
  renderMediaUI();
  showToast('Данные стримера сохранены!', 'success');
}

function renderAdminSqlUsersTable() {
  const tbody = document.getElementById('sql-users-tbody');
  if (!tbody) return;
  tbody.innerHTML = '';
  sqlUsersData.forEach(u => {
    const tr = document.createElement('tr');
    tr.className = 'hover:bg-white/5 transition';
    tr.innerHTML = `
      <td class="p-2 text-red-400 font-bold">${u.dogtag || '#ABI-50'}</td>
      <td class="p-2 text-white font-tactical font-bold">${u.username}</td>
      <td class="p-2 text-gray-400">${u.email}</td>
      <td class="p-2 text-yellow-300 font-bold">${u.role}</td>
      <td class="p-2 text-right"><button onclick="deleteSqlUser('${u.id}')" class="p-1 rounded bg-red-955 text-red-400"><i class="fa-solid fa-trash-can"></i></button></td>
    `;
    tbody.appendChild(tr);
  });
}

function deleteSqlUser(id) {
  if (confirm('Удалить жетон пользователя из базы?')) {
    sqlUsersData = sqlUsersData.filter(x => x.id !== id);
    saveSqlUsersData();
    renderAdminSqlUsersTable();
    showToast('Пользователь удален из базы.', 'info');
  }
}

function adminLogout() {
  isAdminUnlocked = false;
  document.getElementById('admin-locked-view').classList.remove('hidden');
  document.getElementById('admin-unlocked-view').classList.add('hidden');
}

function switchAdminTab(subTab) {
  ['general', 'media', 'discord', 'sql'].forEach(k => {
    const el = document.getElementById('adm-subtab-' + k);
    const btn = document.getElementById('adm-tab-' + k + '-btn');
    if (el) el.classList.add('hidden');
    if (btn) btn.className = 'adm-subtab-btn pb-3 px-2 border-b-2 border-transparent text-gray-400 font-bold';
  });
  const target = document.getElementById('adm-subtab-' + subTab);
  const targetBtn = document.getElementById('adm-tab-' + subTab + '-btn');
  if (target) target.classList.remove('hidden');
  if (targetBtn) targetBtn.className = 'adm-subtab-btn pb-3 px-2 border-b-2 border-red-600 text-white font-bold';
}

function showToast(msg, type = 'info') {
  const container = document.getElementById('toast-container');
  if (!container) return;
  const t = document.createElement('div');
  t.className = 'px-4 py-2.5 rounded-xl border backdrop-blur-md text-xs font-tactical shadow-lg ' + (type === 'success' ? 'bg-emerald-950 border-emerald-600 text-emerald-200' : type === 'error' ? 'bg-red-955 border-red-600 text-red-200' : 'bg-black/90 border-red-800 text-gray-200');
  t.textContent = msg;
  container.appendChild(t);
  setTimeout(() => t.remove(), 3200);
}
